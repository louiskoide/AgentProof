const countdown = useMemo(() => {
  const open = new Date('2026-06-15T00:00:00');
  const diff = open - now;
  if (diff <= 0) return 'WINDOW OPEN';
  const d = Math.floor(diff / 86400000);
  const h = Math.floor((diff % 86400000) / 3600000);
  return { d, h };
}, [now]);
